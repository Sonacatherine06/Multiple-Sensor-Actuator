int ldr=A0;
int pir=2;
int led=9;
int buzzer=8;

void setup()
{
 pinMode(pir,INPUT);
 pinMode(led,OUTPUT);
 pinMode(buzzer,OUTPUT);
}

void loop()
{
 int light=analogRead(ldr);
 int motion=digitalRead(pir);

 if(light<500)
   digitalWrite(led,HIGH);
 else
   digitalWrite(led,LOW);

 if(motion==HIGH)
 {
   digitalWrite(buzzer,HIGH);
   delay(500);
 }
 else
   digitalWrite(buzzer,LOW);
}
